#### 1.2.3 LLM 适配层（`engine/llm_client.py`）

封装 DeepSeek API 调用，基于 OpenAI 兼容 SDK，提供统一接口：

```python
from openai import AsyncOpenAI

class LLMClient:
    """DeepSeek API 封装，内置重试与限流"""

    def __init__(self, api_key: str, base_url: str = "https://api.deepseek.com"):
        self.client = AsyncOpenAI(api_key=api_key, base_url=base_url)
        self.semaphore = asyncio.Semaphore(5)  # 并发控制：最多 5 个同时请求
        self.model = "deepseek-chat"           # 默认模型
        self.max_tokens = 65536                # 输出上限 ~65K

    async def chat(self, system_prompt: str, user_content: str,
                   temperature: float = 0.7, max_retries: int = 3) -> str:
        """
        发送 Chat Completion 请求，内置指数退避重试
        """
        for attempt in range(max_retries):
            try:
                async with self.semaphore:
                    response = await self.client.chat.completions.create(
                        model=self.model,
                        messages=[
                            {"role": "system", "content": system_prompt},
                            {"role": "user", "content": user_content},
                        ],
                        temperature=temperature,
                        max_tokens=self.max_tokens,
                    )
                return response.choices[0].message.content
            except RateLimitError:
                wait = 2 ** attempt  # 1s, 2s, 4s
                await asyncio.sleep(wait)
            except (APITimeoutError, APIConnectionError):
                if attempt == max_retries - 1:
                    raise
                await asyncio.sleep(2 ** attempt)
        raise LLMException("Max retries exceeded for LLM call")
```

**摘要生成策略**：

```python
SUMMARY_PROMPT = """
你是一个小说改编剧本的进度追踪助手。请根据刚完成的剧本片段，提取以下信息：

1. 当前活跃角色及其状态变化（情绪、位置、关系变化）
2. 本段发生的关键情节事件（1-3句话概括）
3. 未解决的伏笔或悬念（如有）

要求：总字数不超过300字，纯中文，无格式标记。
"""

class ContextManager:
    MAX_SUMMARY_CHARS = 300  # 约 400 Token

    async def generate_summary(self, llm_client: LLMClient,
                               fragment_yaml: str) -> str:
        """每次章节转换后，调用轻量 LLM 生成摘要"""
        return await llm_client.chat(
            system_prompt=SUMMARY_PROMPT,
            user_content=fragment_yaml,
            temperature=0.3,  # 低温度保证摘要稳定
            max_retries=2,
        )

    def merge_summaries(self, prev_summary: str,
                        new_summary: str) -> str:
        """合并累积摘要：前文摘要 + 新摘要，总字数控制在 500 字以内"""
        merged = f"【前文摘要】{prev_summary}\n【本章新增】{new_summary}"
        if len(merged) > 500:
            merged = prev_summary[-200:] + "\n" + new_summary  # 截断旧摘要尾部
        return merged
```

#### 1.2.6 剧本拼装器（`engine/script_assembler.py`）

```python
import yaml
from jsonschema import validate, ValidationError

class ScriptAssembler:
    """将各章节片段合并为完整 YAML 剧本，并校验 Schema"""

    def assemble(self, fragments: list[str], script_type: str,
                 novel_title: str, characters: list[dict]) -> str:
        """
        输入: 各章节转换结果 (YAML 片段字符串列表)
        输出: 完整的 YAML 剧本字符串
        """
        # 1. 合并角色表（去重，按首次出场排序）
        merged_characters = self._merge_characters(characters)

        # 2. 按剧本类型组装结构
        if script_type == "movie":
            script_body = self._assemble_movie(fragments)
        elif script_type == "tv_series":
            script_body = self._assemble_tv_series(fragments)
        elif script_type == "mini_series":
            script_body = self._assemble_mini_series(fragments)
        else:
            raise ValueError(f"Unknown script_type: {script_type}")

        # 3. 构建完整 YAML
        yaml_str = yaml.dump({
            "meta": {
                "title": "",  # 用户编辑时填写
                "source_novel": novel_title,
                "script_type": script_type,
                "created_at": datetime.utcnow().isoformat() + "Z",
                "version": 1,
            },
            "characters": merged_characters,
            "script": script_body,
        }, allow_unicode=True, sort_keys=False, default_flow_style=False)

        # 4. Schema 校验
        self._validate_schema(yaml.safe_load(yaml_str), script_type)

        return yaml_str

    def _validate_schema(self, data: dict, script_type: str):
        """校验输出是否合规，不合规抛出 ValidationError"""
        schema = SCHEMA_MAP[script_type]
        validate(instance=data, schema=schema)

    def _assemble_movie(self, fragments: list[str]) -> dict:
        """电影模式：将所有片段场景合并为三幕结构"""
        parsed_fragments = [yaml.safe_load(f) for f in fragments]
        all_scenes = []
        for pf in parsed_fragments:
            if "script" in pf:
                for act in pf["script"].get("acts", []):
                    all_scenes.extend(act.get("scenes", []))
        # 按场景数量均分到三幕
        total = len(all_scenes)
        return {
            "acts": [
                {"act": 1, "title": "建置", "scenes": all_scenes[:total//3]},
                {"act": 2, "title": "对抗", "scenes": all_scenes[total//3:2*total//3]},
                {"act": 3, "title": "结局", "scenes": all_scenes[2*total//3:]},
            ]
        }
```

```python
async def start_conversion(project_id: str, script_type: str, db: AsyncSession):
    # 1. 查询章节列表
    chapters = await db.execute(
        select(Chapter).where(Chapter.project_id == project_id)
                       .order_by(Chapter.chapter_index)
    )
    chapters = chapters.scalars().all()

    # 2. 创建 conversion_tasks 记录
    for ch in chapters:
        task = ConversionTask(
            id=generate_id(),
            project_id=project_id,
            chapter_id=ch.id,
            status="pending",
        )
        db.add(task)

    # 3. 更新项目状态
    project = await db.get(Project, project_id)
    project.status = "converting"
    project.total_chapters = len(chapters)
    await db.commit()

    # 4. 投递 Celery Chord
    launch_conversion(project_id, [ch.id for ch in chapters], script_type)
```

```python
@app.task(bind=True, name="convert_chapter")
def convert_chapter_task(self, chapter_id: str, project_id: str,
                         context_summary: str, script_type: str):
    db = sync_session()
    try:
        chapter = db.query(Chapter).get(chapter_id)
        task = db.query(ConversionTask).filter_by(chapter_id=chapter_id).first()

        task.status = "started"
        task.started_at = datetime.utcnow()
        db.commit()

        # 构建 Prompt
        system_prompt = prompt_builder.build_system(script_type)
        user_content = prompt_builder.build_user(
            context_summary=context_summary,
            chapter_content=chapter.content,
            script_type=script_type,
        )

        # 调用 LLM
        llm = LLMClient(api_key=settings.DEEPSEEK_API_KEY)
        result = asyncio.run(llm.chat(system_prompt, user_content))

        # 生成摘要
        ctx_mgr = ContextManager()
        summary = asyncio.run(ctx_mgr.generate_summary(llm, result))

        # 存储片段
        fragment = ScriptFragment(
            id=generate_id(),
            project_id=project_id,
            chapter_id=chapter_id,
            content_yaml=result,
            summary=summary,
            is_valid=True,
        )
        db.add(fragment)

        task.status = "success"
        task.completed_at = datetime.utcnow()
        db.commit()

        # 推送进度
        publish_progress(project_id, chapter.chapter_index)

        # 返回上下文给 Chord（用于传递给下一章）
        return {
            "chapter_id": chapter_id,
            "fragment_yaml": result,
            "summary": summary,
            "chapter_index": chapter.chapter_index,
        }

    except Exception as e:
        task.status = "failed"
        task.error_message = str(e)
        db.commit()
        raise
    finally:
        db.close()
```

### 4.1 前情摘要生成策略

```
┌─────────────────────────────────────────────────────────────────┐
│                    上下文摘要生命周期                             │
│                                                                 │
│  第1章转换                                                       │
│    │ 输入: System Prompt + 第1章内容（无摘要）                     │
│    │ 输出: 剧本片段1 + 摘要1                                      │
│    ▼                                                            │
│  第2章转换                                                       │
│    │ 输入: System Prompt + 摘要1 + 第2章内容                      │
│    │ 输出: 剧本片段2 + 摘要2 (摘要1合并)                           │
│    ▼                                                            │
│  第3章转换                                                       │
│    │ 输入: System Prompt + 摘要1+2 + 第3章内容                    │
│    │ 输出: 剧本片段3 + 摘要3 (累积)                                │
│    ▼                                                            │
│   ...依此类推                                                     │
└─────────────────────────────────────────────────────────────────┘
```

**摘要提取规则**（写给 LLM 的指令）：

```
你是小说改编剧本的上下文追踪器。请从刚转换的剧本片段中提取：

1. 活跃角色状态：
   - 角色名 + 当前状态（情绪/位置/目标）
   - 仅列出本章有变化的角色，无变化则写"无变化"

2. 关键情节节点（≤3条）：
   - 时间 + 地点 + 事件一句话概括

3. 未解伏笔/悬念：
   - 如有则列出，无则写"无"

输出格式：纯文本，300字以内，无编号无标记。
```

**摘要 Token 预算**：

| 阶段 | 摘要长度 | 约 Token |
|------|---------|---------|
| 单章摘要 | ≤ 300 字 | ~400 Token |
| 累积摘要（前 N-1 章） | ≤ 500 字 | ~700 Token |
| 注入 System Prompt 后 | 约 700 Token | 在 2K 预算内 |

### 4.2 Prompt 模板设计

#### 4.2.1 System Prompt（通用框架）

```
你是一位资深影视编剧，擅长将小说改编为{script_type_name}剧本。

## 输出要求
- 输出格式：严格的 YAML，符合以下 Schema
{schema_description}

## 改编原则
{adaptation_rules}

## 角色一致性
- 保持角色性格和行为逻辑前后一致
- 对白风格需符合角色设定
- 新出场角色需在 characters 列表中声明

## 输出语言
- 中文
- 动作描述使用现在时态
- 对白使用口语化表达
```

#### 4.2.2 三种模式的改编规则

| 模式 | 改编规则（由 `{adaptation_rules}` 注入） |
|------|----------------------------------------|
| **电影** | 三幕结构（建置/对抗/结局）；每场控制在1-3分钟阅读量；场景数量60-120场；聚焦主线，删除次要支线 |
| **电视剧** | 按集划分，每集包含1-2幕；每集结尾设置悬念钩子；多线叙事可保留；每集10-15场 |
| **微短剧** | 单集1-3分钟，5-8场；每场必须标注`duration_seconds`；每集开头3秒内进入冲突；每集结尾必须有`hook`；台词每句不超过15字；场景切换频繁 |

#### 4.2.3 User Message 结构

```
{context_block}

## 当前章节: {chapter_title}

{chapter_content}

请将以上章节内容转换为{script_type_name}格式的YAML片段。
```

```python
class PromptBuilder:
    def build_system(self, script_type: str) -> str:
        rules = {
            "movie": "三幕结构...",
            "tv_series": "按集划分...",
            "mini_series": "单集1-3分钟...",
        }
        schema = SCHEMA_TEMPLATES[script_type]  # YAML Schema 模板字符串
        return SYSTEM_PROMPT_TEMPLATE.format(
            script_type_name=SCRIPT_TYPE_NAMES[script_type],
            schema_description=schema,
            adaptation_rules=rules[script_type],
        )

    def build_user(self, context_summary: str, chapter_content: str,
                   script_type: str, chapter_title: str) -> str:
        context_block = ""
        if context_summary:
            context_block = (
                "## 前情摘要（请严格继承以下设定）\n"
                f"{context_summary}\n"
            )
        return f"""{context_block}
## 当前章节: {chapter_title}

{chapter_content}

## 输出指令
请将以上章节内容转换为{SCRIPT_TYPE_NAMES[script_type]}格式的YAML片段。
"""
```

### 5.2 章节级失败不阻断整体

```python
@app.task(bind=True, name="convert_chapter", max_retries=2,
          default_retry_delay=5)
def convert_chapter_task(self, chapter_id: str, project_id: str,
                         context_summary: str, script_type: str):
    try:
        # ... 正常转换逻辑 ...
        return {"chapter_id": chapter_id, "status": "success", ...}

    except SoftTimeLimitExceeded:
        # 超时：标记失败，不阻断
        return _mark_chapter_failed(chapter_id, project_id, "转换超时(>9分钟)")

    except Exception as e:
        # 自动重试（Celery 内置机制）
        try:
            raise self.retry(exc=e, countdown=5 * (2 ** self.request.retries))
        except self.MaxRetriesExceededError:
            # 重试次数耗尽，标记失败但不阻断整体
            return _mark_chapter_failed(chapter_id, project_id, str(e))

def _mark_chapter_failed(chapter_id: str, project_id: str,
                         error_msg: str) -> dict:
    """标记章节失败，返回空片段，merge任务会跳过"""
    db = sync_session()
    try:
        task = db.query(ConversionTask).filter_by(chapter_id=chapter_id).first()
        task.status = "failed"
        task.error_message = error_msg
        chapter = db.query(Chapter).get(chapter_id)
        chapter.status = "failed"
        db.commit()
        publish_progress(project_id, chapter.chapter_index, "failed")
    finally:
        db.close()
    return {
        "chapter_id": chapter_id,
        "status": "failed",
        "fragment_yaml": "",  # 空片段，merge时跳过
        "summary": "",
    }
```

### 5.3 合并时的失败处理

```python
@app.task(bind=True, name="merge_scripts")
def merge_scripts_task(self, results: list[dict], project_id: str,
                       script_type: str):
    db = sync_session()
    try:
        project = db.query(Project).get(project_id)

        # 分离成功和失败的片段
        success = [r for r in results if r.get("status") == "success"]
        failed = [r for r in results if r.get("status") == "failed"]

        if not success:
            # 全部失败
            project.status = "failed"
            project.error_message = f"所有{len(results)}个章节转换均失败"
            db.commit()
            publish_progress(project_id, "failed", project.progress)
            return

        # 合并成功的片段
        fragments = [r["fragment_yaml"] for r in success]
        assembler = ScriptAssembler()
        yaml_output = assembler.assemble(fragments, script_type, ...)

        project.script_yaml = yaml_output
        project.status = "completed" if not failed else "completed"
        if failed:
            project.error_message = (
                f"转换完成，但 {len(failed)} 个章节转换失败，"
                f"已跳过: {', '.join(r['chapter_id'] for r in failed)}"
            )
        project.progress = 100
        db.commit()
        publish_progress(project_id, "completed", 100)

    except Exception as e:
        project.status = "failed"
        project.error_message = f"合并剧本时发生错误: {str(e)}"
        db.commit()
        publish_progress(project_id, "failed", project.progress)
    finally:
        db.close()
```

## 6. WebSocket 进度推送实现

### 6.1 架构设计

```
┌──────────┐  WebSocket   ┌──────────┐  Redis Pub/Sub  ┌──────────┐
│  Browser │◀────────────▶│ FastAPI  │◀───────────────▶│  Redis   │
│          │  /ws/projects│ (WS Mgr) │                  │ Pub/Sub  │
│          │  /{id}/progress│         │                  │          │
└──────────┘              └──────────┘                  └──────────┘
                                                              ▲
                                                              │
                                                    ┌─────────┴────────┐
                                                    │  Celery Worker   │
                                                    │  PUBLISH         │
                                                    │  progress:{pid}  │
                                                    └──────────────────┘
```

**设计要点**：
- Celery Worker 不直接与 WebSocket 通信，而是通过 **Redis Pub/Sub** 作为中间层解耦
- FastAPI 的 WebSocket Manager 订阅 Redis 频道，收到消息后转发给对应客户端
- 频道命名规则：`progress:{project_id}`

### 6.2 WebSocket 管理器（`services/progress_service.py`）

```python
import json
import asyncio
from fastapi import WebSocket
from redis.asyncio import Redis

class ProgressManager:
    """WebSocket 连接管理器 + Redis 订阅"""

    def __init__(self, redis: Redis):
        self.redis = redis
        # project_id → set of WebSocket connections
        self._connections: dict[str, set[WebSocket]] = {}

    async def connect(self, project_id: str, websocket: WebSocket):
        await websocket.accept()
        if project_id not in self._connections:
            self._connections[project_id] = set()
        self._connections[project_id].add(websocket)

    async def disconnect(self, project_id: str, websocket: WebSocket):
        if project_id in self._connections:
            self._connections[project_id].discard(websocket)
            if not self._connections[project_id]:
                del self._connections[project_id]

    async def broadcast(self, project_id: str, message: dict):
        """向所有订阅该项目的客户端推送消息"""
        if project_id in self._connections:
            dead = []
            for ws in self._connections[project_id]:
                try:
                    await ws.send_json(message)
                except Exception:
                    dead.append(ws)
            for ws in dead:
                self._connections[project_id].discard(ws)

    async def start_listener(self):
        """启动 Redis Pub/Sub 监听，将消息转发到 WebSocket"""
        pubsub = self.redis.pubsub()
        await pubsub.psubscribe("progress:*")

        async for msg in pubsub.listen():
            if msg["type"] != "pmessage":
                continue
            channel = msg["channel"].decode()
            project_id = channel.split(":", 1)[1]
            data = json.loads(msg["data"])
            await self.broadcast(project_id, data)
```

```python
@router.websocket("/ws/projects/{project_id}/progress")
async def progress_websocket(
    project_id: str,
    websocket: WebSocket,
    progress_mgr: ProgressManager = Depends(get_progress_manager),
    current_user: User = Depends(get_current_user_ws),  # Token 通过 query param 传递

    if project.user_id != current_user.id:

        return

    await progress_mgr.connect(project_id, websocket)
    try:
        # 保持连接，等待客户端断开
        while True:
            await websocket.receive_text()  # 心跳检测
    except WebSocketDisconnect:
        await progress_mgr.disconnect(project_id, websocket)
```
